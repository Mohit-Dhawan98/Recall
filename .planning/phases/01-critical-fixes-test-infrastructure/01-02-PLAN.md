```python
from urllib.parse import urlparse, parse_qsl, urlencode, urlunparse

def normalize_url(url: str) -> str:
    """Normalize URL for deduplication.

    - Removes tracking parameters (utm_*, fbclid, gclid, etc.)
    - Sorts query parameters alphabetically for consistent ordering
    - Removes trailing slashes from path
    - Removes URL fragments (#section)
    """
    parsed = urlparse(url)

    # Tracking parameters to remove
    tracking_params = {
        'utm_source', 'utm_medium', 'utm_campaign', 'utm_term', 'utm_content',
        'ref', 'fbclid', 'gclid', 'source', 'mc_cid', 'mc_eid'
    }

    if parsed.query:
        # Parse query string to list of (key, value) tuples
        params = parse_qsl(parsed.query)

        # Filter out tracking parameters
        filtered_params = [
            (k, v) for k, v in params
            if k not in tracking_params
        ]

        # Sort by key, then by value for consistent ordering
        sorted_params = sorted(filtered_params)

        # Rebuild query string
        query = urlencode(sorted_params)
    else:
        query = ''

    # Reconstruct URL without fragment, with trailing slash removed
    normalized = urlunparse((
        parsed.scheme,
        parsed.netloc,
        parsed.path.rstrip('/'),  # Remove trailing slash
        parsed.params,            # URL params (rarely used, e.g., ;type=a)
        query,                    # Sorted query string
        ''                        # No fragment
    ))

    return normalized
```

Key changes from current implementation:
1. Use `parse_qsl` instead of manual string splitting (handles edge cases like encoded chars)
2. Use `sorted()` on list of tuples (sorts by key, then value)
3. Use `urlencode` to rebuild query (handles encoding properly)
4. Use `urlunparse` to rebuild full URL (preserves all components)
5. Convert tracking_params to a set for O(1) lookup
